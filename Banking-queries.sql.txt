-- Display all customer details - SELECT
SELECT *
FROM cust_info;

-- Display customer names and states - SELECT
SELECT cust_name, cust_state
FROM cust_info;

-- Display customers from Chennai - WHERE
SELECT *
FROM cust_info
WHERE cust_city = 'Chennai';

-- Display accounts with balance greater than 50000 - Greater Than (>)
SELECT acc_id, balance, status
FROM acc_info
WHERE balance > 50000;

-- Display active accounts with balance greater than 20000 - AND
SELECT acc_id, balance, status
FROM acc_info
WHERE balance > 20000
  AND status = 'Active';

-- Display accounts with balance greater than 100000 or inactive status - OR
SELECT acc_id, balance, status
FROM acc_info
WHERE balance > 100000
   OR status = 'InActive';

-- Display accounts that are not active - NOT
SELECT acc_id, balance, status
FROM acc_info
WHERE NOT (status = 'Active');

-- Display customers from Chennai or Mumbai - IN
SELECT cust_id, cust_name, cust_city
FROM cust_info
WHERE cust_city IN ('Chennai', 'Mumbai');

-- Display customers who are not from Chennai or Mumbai - NOT IN
SELECT cust_id, cust_name, cust_city
FROM cust_info
WHERE cust_city NOT IN ('Chennai', 'Mumbai');

-- Display accounts with balances between 20000 and 80000 - BETWEEN
SELECT acc_id, balance, status
FROM acc_info
WHERE balance BETWEEN 20000 AND 80000;

-- Display customers whose city information is missing - IS NULL
SELECT cust_id, cust_name, cust_city
FROM cust_info
WHERE cust_city IS NULL;

-- Display customers whose city information is available - IS NOT NULL
SELECT cust_id, cust_name, cust_city
FROM cust_info
WHERE cust_city IS NOT NULL;

-- Display customers in alphabetical order - ORDER BY ASC
SELECT cust_id, cust_name, cust_city
FROM cust_info
ORDER BY cust_name ASC;

-- Display the top 3 accounts with the highest balances - ORDER BY and LIMIT
SELECT acc_id, balance, status
FROM acc_info
ORDER BY balance DESC
LIMIT 3;

-- Count the total number of bank accounts - COUNT()
SELECT COUNT(*)
FROM acc_info;

-- Calculate the total balance across all bank accounts - SUM()
SELECT SUM(balance)
FROM acc_info;

-- Calculate the average balance across all bank accounts - AVG()
SELECT AVG(balance)
FROM acc_info;

-- Find the lowest account balance - MIN()
SELECT MIN(balance)
FROM acc_info;

-- Find the highest account balance - MAX()
SELECT MAX(balance)
FROM acc_info;

-- Calculate total balance for each account status - GROUP BY
SELECT status, SUM(balance) AS total_balance
FROM acc_info
GROUP BY status;

-- Calculate total balance maintained at each branch - GROUP BY
SELECT branch_id, SUM(balance) AS total_balance
FROM acc_info
GROUP BY branch_id;

-- Display branches with total account balance above 100000 - HAVING
SELECT branch_id, SUM(balance) AS total_balance
FROM acc_info
GROUP BY branch_id
HAVING total_balance > 100000;

-- Display the top 3 branches by total account balance - GROUP BY, ORDER BY and LIMIT
SELECT branch_id, SUM(balance) AS total_balance
FROM acc_info
GROUP BY branch_id
ORDER BY total_balance DESC
LIMIT 3;

-- Display customer names with their account IDs - INNER JOIN
SELECT c.cust_name, a.acc_id
FROM cust_info c
INNER JOIN acc_holders a
    ON c.cust_id = a.cust_id;

-- Display customer names with account IDs and balances - INNER JOIN
SELECT c.cust_name, a.acc_id, b.balance
FROM cust_info c
INNER JOIN acc_holders a
    ON c.cust_id = a.cust_id
INNER JOIN acc_info b
    ON b.acc_id = a.acc_id;

-- Display all customers with their account IDs, including customers without accounts - LEFT JOIN
SELECT c.cust_name, a.acc_id
FROM cust_info c
LEFT JOIN acc_holders a
    ON c.cust_id = a.cust_id;

-- Display customers who do not have any bank account - LEFT JOIN
SELECT c.cust_id, c.cust_name
FROM cust_info c
LEFT JOIN acc_holders a
    ON c.cust_id = a.cust_id
WHERE a.acc_id IS NULL;

-- Display all branches with their loan details, including branches without loans - RIGHT JOIN
SELECT b.branch_name, l.loan_id, l.loan_type
FROM loans l
RIGHT JOIN branch b
    ON b.branch_id = l.branch_id;

-- Display all transaction channels with their transactions, including unused channels - RIGHT JOIN
SELECT a.channel_name, b.trans_id, b.amount, b.trans_status
FROM transxs b
RIGHT JOIN trans_channel a
    ON a.channel_id = b.channel_id;

-- Combine customer cities and branch states into one location list - UNION
SELECT cust_city AS location
FROM cust_info
UNION
SELECT branch_state
FROM branch;

-- Combine account holders and loan customers without duplicate IDs - UNION
SELECT cust_id
FROM acc_holders
UNION
SELECT cust_id
FROM loans;

-- Combine account holders and loan customers while retaining duplicates - UNION ALL
SELECT cust_id
FROM acc_holders
UNION ALL
SELECT cust_id
FROM loans;

-- Display accounts with balance greater than the overall average - Subquery
SELECT acc_id, balance
FROM acc_info
WHERE balance > (
    SELECT AVG(balance)
    FROM acc_info
);

-- Display the customer with the highest loan principal - Subquery
SELECT c.cust_name, l.loan_principal
FROM cust_info c
INNER JOIN loans l
    ON c.cust_id = l.cust_id
WHERE l.loan_principal = (
    SELECT MAX(loan_principal)
    FROM loans
);

-- Display loans with principal greater than at least one Personal loan - ANY
SELECT loan_id, loan_type, loan_principal
FROM loans
WHERE loan_principal > ANY (
    SELECT loan_principal
    FROM loans
    WHERE loan_type = 'Personal'
);

-- Display branches with total account balance greater than 100000 - Temporary Table
SELECT branch_id, total_balance
FROM (
    SELECT branch_id, SUM(balance) AS total_balance
    FROM acc_info
    GROUP BY branch_id
) AS branch_summary
WHERE total_balance > 100000;

-- Display accounts with balance greater than 50000 - CTE
WITH high_balance_accounts AS (
    SELECT acc_id, branch_id, balance
    FROM acc_info
    WHERE balance > 50000
)
SELECT *
FROM high_balance_accounts;

-- Display every transaction with the overall average transaction amount - Window Function OVER()
SELECT trans_id, acc_id, amount,
       AVG(amount) OVER() AS average_amount
FROM transxs;

-- Display every transaction with its account-wise total transaction amount - PARTITION BY
SELECT trans_id, acc_id, amount,
       SUM(amount) OVER(PARTITION BY acc_id) AS account_total
FROM transxs;

-- Rank loan payments by EMI amount from highest to lowest - RANK
SELECT payment_id, loan_id, emi_amount,
       RANK() OVER(ORDER BY emi_amount DESC) AS payment_rank
FROM loan_payments;

-- Assign unique serial numbers to loan payments by EMI amount - ROW_NUMBER
SELECT payment_id, loan_id, emi_amount,
       ROW_NUMBER() OVER(ORDER BY emi_amount DESC) AS serial_number
FROM loan_payments;

-- Display transaction channel names in uppercase, lowercase and combined format - String Functions
SELECT channel_id, channel_name,
       UPPER(channel_name) AS uppercase,
       LOWER(channel_name) AS lowercase,
       CONCAT(channel_name, ' ', 'channel') AS combined
FROM trans_channel;

-- Display transaction channel name length and first four characters - String Functions
SELECT
    channel_id,
    channel_name,
    LENGTH(channel_name) AS no_of_characters,
    SUBSTRING(channel_name, 1, 4) AS first_four_characters
FROM trans_channel;

-- Find the position of letter a in transaction channel names - INSTR and LOCATE
SELECT
    channel_id,
    channel_name,
    INSTR(channel_name, 'a') AS instr_position,
    LOCATE('a', channel_name) AS locate_position
FROM trans_channel;

-- Display cards whose card type starts with D - LIKE
SELECT card_id, acc_id, card_type, card_status
FROM cards
WHERE card_type LIKE 'D%';

-- Display loan payment dates in DD-MM-YYYY format - DATE_FORMAT
SELECT
    payment_id,
    loan_id,
    payment_date,
    DATE_FORMAT(payment_date, '%d-%m-%Y') AS formatted_date
FROM loan_payments;

-- Display loan payments made within the last 30 days - INTERVAL
SELECT payment_id, loan_id, payment_date, emi_amount
FROM loan_payments
WHERE payment_date >= CURDATE() - INTERVAL 30 DAY;
